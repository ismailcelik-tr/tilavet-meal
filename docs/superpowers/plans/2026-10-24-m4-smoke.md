```python
"""Meal playback queue: one meal at a time, never cut mid-sentence. Pure; time is passed in."""

from collections import deque
from collections.abc import Callable
from dataclasses import dataclass

from tilavet.corpus import MealUnit
from tilavet.policy import Play

__all__ = ["MealPlayer", "PlayerEvent", "Skipped", "Started"]


@dataclass(frozen=True)
class Started:
    t: float
    play: Play


@dataclass(frozen=True)
class Skipped:
    t: float
    play: Play


PlayerEvent = Started | Skipped


class MealPlayer:
    def __init__(self, duration_s: Callable[[MealUnit], float], max_wait_s: float) -> None:
        self._duration_s = duration_s
        self._max_wait_s = max_wait_s
        self._queue: deque[Play] = deque()
        self._free_at = float("-inf")

    def submit(self, play: Play) -> list[PlayerEvent]:
        # Settle earlier meals first: whether a newer one exists is judged at their start time.
        events = self.advance(play.t)
        self._queue.append(play)
        return events + self.advance(play.t)

    def advance(self, now: float) -> list[PlayerEvent]:
        events: list[PlayerEvent] = []
        while self._queue:
            play = self._queue[0]
            start = max(self._free_at, play.t)
            if start > now:
                break

            self._queue.popleft()
            if self._queue and start - play.t > self._max_wait_s:
                events.append(Skipped(start, play))
                continue

            events.append(Started(start, play))
            self._free_at = start + self._duration_s(play.unit)
        return events
```

```python
"""Where a played signal starts inside its recording (speaker -> microphone latency)."""

import numpy as np

__all__ = ["estimate_offset"]


def estimate_offset(reference: np.ndarray, recording: np.ndarray, sample_rate: int, max_lag_s: float) -> float:
    """Lag in [0, max_lag_s] seconds that best aligns `reference` with `recording` (FFT cross-correlation)."""
    size = 1 << (len(reference) + len(recording) - 1).bit_length()
    spectrum = np.fft.rfft(recording, size) * np.conj(np.fft.rfft(reference, size))
    correlation = np.fft.irfft(spectrum, size)[: round(max_lag_s * sample_rate) + 1]
    return int(np.argmax(correlation)) / sample_rate
```

```python
"""Live pipeline over microphone chunks: window -> ASR -> engine -> policy -> player. Pure; time is passed in."""

from collections.abc import Callable, Sequence
from dataclasses import dataclass
from typing import Protocol

import numpy as np

from tilavet.asr import Recognizer
from tilavet.audio import SAMPLE_RATE
from tilavet.corpus import normalize
from tilavet.engine import Event
from tilavet.live.player import MealPlayer, PlayerEvent
from tilavet.policy import EmissionPolicy, Play

__all__ = ["LiveSession", "Record", "Window"]


class Engine(Protocol):
    def feed(self, t: float, words: Sequence[str]) -> list[Event]: ...


@dataclass(frozen=True)
class Window:
    heard_s: float  # audio heard when the window closed
    t: float  # decision time: heard_s + queueing + compute
    text: str
    skipped: int  # hops dropped because ASR fell behind


Record = Window | Event | Play | PlayerEvent


class LiveSession:
    def __init__(
        self,
        recognizer: Recognizer,
        engine: Engine,
        policy: EmissionPolicy,
        player: MealPlayer,
        clock: Callable[[], float],
        hop_s: float,
        window_s: float,
    ) -> None:
        self._recognizer = recognizer
        self._engine = engine
        self._policy = policy
        self._player = player
        self._clock = clock
        self._hop = round(hop_s * SAMPLE_RATE)
        self._window = round(window_s * SAMPLE_RATE)
        self._buffer = np.zeros(0, dtype=np.float32)
        self._heard = 0
        self._next_hop = self._hop

    def feed(self, samples: np.ndarray, arrived_at: float) -> list[Record]:
        """`arrived_at`: clock time when the last of `samples` was captured."""
        self._buffer = np.concatenate([self._buffer, samples])[-self._window :]
        self._heard += len(samples)
        if self._heard < self._next_hop:
            return []

        hops = (self._heard - self._next_hop) // self._hop + 1
        self._next_hop += hops * self._hop
        heard_s = self._heard / SAMPLE_RATE

        text = self._recognizer.transcribe(self._buffer)
        t = heard_s + self._clock() - arrived_at
        records: list[Record] = [Window(heard_s, t, text, hops - 1)]

        for event in self._engine.feed(t, normalize(text).split()):
            records.append(event)
            play = self._policy.on(event)
            if play:
                records.append(play)
                records.extend(self._player.submit(play))

        records.extend(self._player.advance(t))
        return records
```

```python
"""sounddevice adapters. Devices are picked by name, never by system default."""

import queue
import time
import unicodedata
from enum import Enum

import numpy as np
import sounddevice as sd

from tilavet.audio import SAMPLE_RATE

__all__ = ["Kind", "Microphone", "default_output", "find_device", "play"]


class Kind(Enum):
    INPUT = "input"
    OUTPUT = "output"


def _nfc(name: str) -> str:
    # CoreAudio may report "Hoparlörü" decomposed (o + combining diaeresis).
    return unicodedata.normalize("NFC", name)


def find_device(name: str, kind: Kind) -> int:
    for index, device in enumerate(sd.query_devices()):
        if _nfc(device["name"]) == _nfc(name) and device[f"max_{kind.value}_channels"] > 0:
            return index
    raise LookupError(f"no {kind.value} device named {name!r}")


def default_output() -> str:
    return _nfc(sd.query_devices(kind="output")["name"])


class Microphone:
    """16 kHz mono chunks stamped with arrival time; counts overflows (lost samples)."""

    def __init__(self, name: str) -> None:
        self.chunks: queue.Queue[tuple[np.ndarray, float]] = queue.Queue()
        self.overflows = 0
        self._stream = sd.InputStream(
            device=find_device(name, Kind.INPUT),
            samplerate=SAMPLE_RATE,
            channels=1,
            dtype="float32",
            callback=self._callback,
        )

    def _callback(self, data: np.ndarray, frames: int, time_info: object, status: sd.CallbackFlags) -> None:
        if status.input_overflow:
            self.overflows += 1
        self.chunks.put((data[:, 0].copy(), time.monotonic()))

    def __enter__(self) -> "Microphone":
        self._stream.start()
        return self

    def __exit__(self, *exc: object) -> None:
        self._stream.stop()
        self._stream.close()


def play(audio: np.ndarray, device: str) -> None:
    sd.play(audio, SAMPLE_RATE, device=find_device(device, Kind.OUTPUT), blocking=True)
```
